水：ポスト作成　木：ポスト添削　金：投稿＆交流

/**
 * FIREセミナーGoogleフォーム（固定IDで更新）
 */
function runFireForm() {
  const STRIP_EMOJI = true;
  const FORM_ID_KEY = "FIRE_FORM_ID";
  const title = "FIREセミナー予約";

  const shortDescription =
`こちらのフォームからご予約してください。
セミナーに参加される際は、顔出し・ミュートでご参加です。
最初から最後まで参加できる日時で予約お願いいたします。
必ず下記の動画をみて予約してください！
新井さんの2分50秒動画
https://leapsfire.hp.peraichi.com`;

  const longInfo =
`☝️LINKの仕組みが素晴らしく稼げて広がる理由が分かります。
＜説明会zoom日程＞
新井さん×ヤッシーFIREセミナー / オンラインzoom開催 / 参加費 無料
1/15(木)21:00～ / 1/22(木)21:00～ / 1/24(土)21:00～
1/29(木)21:00～ / 1/31(土)21:00～
必ず来られる方だけ予約してください。
キャンセルされたい場合は事前にご連絡が必須です。
LINKに入った場合は私もしっかりサポートします。
LINKセミナーに必ず参加してください。`;

  const sanitize = (s) => {
    if (!STRIP_EMOJI) return String(s);
    return String(s).replace(/[\uD800-\uDBFF][\uDC00-\uDFFF]/g, "");
  };

  const props = PropertiesService.getScriptProperties();
  const savedId = props.getProperty(FORM_ID_KEY);
  let form = null, created = false;

  if (savedId) {
    try { form = FormApp.openById(savedId); } catch (e) { form = null; }
  }
  if (!form) {
    form = FormApp.create(sanitize(title));
    props.setProperty(FORM_ID_KEY, form.getId());
    created = true;
  }

  form.setTitle(sanitize(title));
  try { form.setDescription(sanitize(shortDescription)); } catch (e) {}
  resetFormItems_(form);

  form.addSectionHeaderItem().setTitle("FIREセミナー").setHelpText(sanitize(longInfo));
  form.addTextItem().setTitle("オプチャ名").setRequired(true);
  form.addTextItem().setTitle("お名前（フルネーム・カタカナ）").setRequired(true);

  const dateChoices = [
    "1/15(木)21:00～","1/22(木)21:00～","1/24(土)21:00～",
    "1/29(木)21:00～","1/31(土)21:00～"
  ].map(sanitize);

  form.addMultipleChoiceItem().setTitle("参加希望日").setChoiceValues(dateChoices).setRequired(true);

  Logger.log(created ? "CREATED" : "UPDATED");
  Logger.log("Edit URL = " + form.getEditUrl());
  Logger.log("Public URL = " + form.getPublishedUrl());
}

function resetFormItems_(form) {
  const items = form.getItems();
  for (let i = items.length - 1; i >= 0; i--) form.deleteItem(items[i]);
}
